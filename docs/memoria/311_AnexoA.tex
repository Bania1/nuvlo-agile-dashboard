\begin{table}
    \centering
    \scalebox{0.8}{
        \begin{tabular}{ m{8em} m{33em} } 
        \textbf{Caso de Uso} &  Nombre del caso de uso \\
        \hline
        \textbf{Actores} & Usuario no registrado. \\
        \hline
        \textbf{Descripción} & El objetivo del caso de uso es que cualquier usuario pueda registrarse para hacer uso de la aplicación. \\
        \hline
        \textbf{Precondición} & El usuario no debe estar previamente registrado en la aplicación. \\  
        \hline
        \textbf{Flujo principal} & 
            \begin{enumerate}
                \item El Caso de Uso comienza cuando el usuario selecciona la opción de registro.
                \item La Aplicación solicita al usuario un email y una contraseña.
                \item El usuario introduce el email.
                \item El usuario introduce la contraseña.
                \item La aplicación valida que el email y la contraseña es correcto.
                \item \textbf{Mientras} la contraseña o email no sean correctos.
                    \begin{enumerate}
                    \item La aplicación solicita al usuario un email.
                    \item El usuario introduce el email.
                    \item La aplicación solicita al usuario una contraseña.
                    \item El usuario introduce la contraseña.
                    \item La aplicación valida los datos son correctos.
                    \end{enumerate}
                \item El usuario selecciona la opción de finalizar registro
                \item La aplicación informa al usuario que se ha registrado correctamente.
            \end{enumerate} \\
        \hline
        \textbf{Postcondición} & El usuario se ha registrado aplicación. \\
        \hline
        \textbf{Flujo Alternativo} & Cancelar: el usuario cancela el registro y vuelve a la página de Inicio. \\
        \hline
        \end{tabular}
    }
